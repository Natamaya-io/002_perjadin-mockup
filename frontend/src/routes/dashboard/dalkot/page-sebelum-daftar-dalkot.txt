<script lang="ts">
    import { onMount } from 'svelte';
    import { goto } from '$app/navigation';
    import {
        CalendarDays,
        ChevronDown,
        ChevronUp,
        CircleDollarSign,
        FileText,
        MapPin,
        Pencil,
        Plus,
        Printer,
        RefreshCcw,
        Search,
        Trash2,
        Users,
        X
    } from 'lucide-svelte';
    import { api } from '$lib/shared/api';
    import { toast } from '$lib/shared/stores/toast';

    type DalkotOfficer = {
        id: string;
        dalkotId: string;
        name: string;
        nip: string;
        position: string;
        amount: number;
        sortOrder: number;
        createdAt: string;
    };

    type DalkotRow = {
        id: string;
        businessId: string;
        activeEmail: string;
        serviceDate: string;
        category: string;
        accompaniedOfficial: string;
        dalkotType: 'SPJ_RILL' | 'KEBIJAKAN_PROTOKOL';
        policyCategory?: string;
        activityName: string;
        location: string;
        reportContent?: string;
        status: 'DRAFT' | 'DIAJUKAN' | 'SELESAI' | 'DIBATALKAN';
        totalAmount: number;
        documentOriginalName?: string;
        documentMime?: string;
        documentSize?: number;
        documentUrl?: string;
        officerCount: number;
        officers: DalkotOfficer[];
        createdAt: string;
        updatedAt: string;
    };

    type DalkotSummary = {
        totalRecords: number;
        totalOfficers: number;
        totalAmount: number;
        pendingRecords: number;
    };

    type DalkotResponse = {
        rows: DalkotRow[];
        summary: DalkotSummary;
    };

    const emptySummary = (): DalkotSummary => ({
        totalRecords: 0,
        totalOfficers: 0,
        totalAmount: 0,
        pendingRecords: 0
    });

    const monthOptions = [
        ['01', 'Januari'],
        ['02', 'Februari'],
        ['03', 'Maret'],
        ['04', 'April'],
        ['05', 'Mei'],
        ['06', 'Juni'],
        ['07', 'Juli'],
        ['08', 'Agustus'],
        ['09', 'September'],
        ['10', 'Oktober'],
        ['11', 'November'],
        ['12', 'Desember']
    ];

    let rows: DalkotRow[] = [];
    let summary: DalkotSummary = emptySummary();
    let loading = true;
    let deleting = false;
    let loadError = '';

    let searchQuery = '';
    let typeFilter = '';
    let monthFilter = '';
    let yearFilter = '';

    let expandedIds = new Set<string>();
    let deleteOpen = false;
    let selectedRow: DalkotRow | null = null;

    $: availableYears = Array.from(
        new Set(
            rows
                .map((row) => row.serviceDate?.slice(0, 4))
                .filter(Boolean)
        )
    ).sort((first, second) => Number(second) - Number(first));

    $: filteredRows = rows.filter((row) => {
        const keyword = searchQuery.trim().toLowerCase();

        const matchSearch =
            keyword === '' ||
            row.businessId?.toLowerCase().includes(keyword) ||
            row.activityName?.toLowerCase().includes(keyword) ||
            row.location?.toLowerCase().includes(keyword) ||
            row.accompaniedOfficial?.toLowerCase().includes(keyword) ||
            row.category?.toLowerCase().includes(keyword) ||
            row.officers?.some((officer) =>
                officer.name?.toLowerCase().includes(keyword)
            );

        const matchType =
            typeFilter === '' ||
            row.dalkotType === typeFilter;

        const matchMonth =
            monthFilter === '' ||
            row.serviceDate?.slice(5, 7) === monthFilter;

        const matchYear =
            yearFilter === '' ||
            row.serviceDate?.slice(0, 4) === yearFilter;

        return matchSearch && matchType && matchMonth && matchYear;
    });

    onMount(loadDalkot);

    async function loadDalkot() {
        loading = true;
        loadError = '';

        try {
            const result = await (api as any).getDalkotList() as DalkotResponse;

            rows = Array.isArray(result?.rows)
                ? result.rows
                : [];

            summary = result?.summary ?? emptySummary();
        } catch (error) {
            loadError =
                error instanceof Error
                    ? error.message
                    : 'Daftar Dalkot gagal dimuat.';

            toast.error(loadError);
        } finally {
            loading = false;
        }
    }

    function resetFilters() {
        searchQuery = '';
        typeFilter = '';
        monthFilter = '';
        yearFilter = '';
    }

    function toggleOfficers(id: string) {
        const next = new Set(expandedIds);

        if (next.has(id)) {
            next.delete(id);
        } else {
            next.add(id);
        }

        expandedIds = next;
    }

    function openDelete(row: DalkotRow) {
        selectedRow = row;
        deleteOpen = true;
    }

    function closeDelete() {
        if (deleting) return;

        deleteOpen = false;
        selectedRow = null;
    }

    async function deleteDalkot() {
        if (!selectedRow) return;

        deleting = true;

        try {
            await (api as any).deleteDalkot(selectedRow.id);
            toast.success('Data Dalkot berhasil dihapus.');
            deleteOpen = false;
            selectedRow = null;
            await loadDalkot();
        } catch (error) {
            toast.error(
                error instanceof Error
                    ? error.message
                    : 'Data Dalkot gagal dihapus.'
            );
        } finally {
            deleting = false;
        }
    }

    function formatRupiah(value: number): string {
        return new Intl.NumberFormat('id-ID', {
            style: 'currency',
            currency: 'IDR',
            maximumFractionDigits: 0
        }).format(Number(value || 0));
    }

    function formatDate(value: string): string {
        if (!value) return '-';

        const normalized =
            value.length === 10
                ? `${value}T00:00:00`
                : value;

        const date = new Date(normalized);

        if (Number.isNaN(date.getTime())) {
            return value;
        }

        return new Intl.DateTimeFormat('id-ID', {
            day: '2-digit',
            month: 'long',
            year: 'numeric'
        }).format(date);
    }

    function formatType(row: DalkotRow): string {
        if (row.dalkotType === 'SPJ_RILL') {
            return 'SPJ RILL';
        }

        return 'Kebijakan Protokol';
    }

    function formatPolicyCategory(value = ''): string {
        const labels: Record<string, string> = {
            JAM_KERJA: 'Jam Kerja',
            OVERTIME: 'Overtime',
            HARI_LIBUR: 'Hari Libur'
        };

        return labels[value] ?? value ?? '-';
    }

    function statusClass(status: DalkotRow['status']): string {
        const classes: Record<DalkotRow['status'], string> = {
            DRAFT: 'border-slate-200 bg-slate-100 text-slate-700',
            DIAJUKAN: 'border-amber-200 bg-amber-50 text-amber-700',
            SELESAI: 'border-emerald-200 bg-emerald-50 text-emerald-700',
            DIBATALKAN: 'border-red-200 bg-red-50 text-red-700'
        };

        return classes[status];
    }

    function statusLabel(status: DalkotRow['status']): string {
        const labels: Record<DalkotRow['status'], string> = {
            DRAFT: 'Draft',
            DIAJUKAN: 'Diajukan',
            SELESAI: 'Selesai',
            DIBATALKAN: 'Dibatalkan'
        };

        return labels[status];
    }

    function escapeHtml(value: unknown): string {
        return String(value ?? '')
            .replaceAll('&', '&amp;')
            .replaceAll('<', '&lt;')
            .replaceAll('>', '&gt;')
            .replaceAll('"', '&quot;')
            .replaceAll("'", '&#039;');
    }

    function openPrintWindow(title: string, body: string) {
        const printWindow = window.open('', '_blank', 'width=980,height=760');

        if (!printWindow) {
            toast.error('Popup cetak diblokir oleh peramban.');
            return;
        }

        printWindow.document.open();
        printWindow.document.write(`
            <!doctype html>
            <html lang="id">
                <head>
                    <meta charset="utf-8" />
                    <title>${escapeHtml(title)}</title>
                    <style>
                        * { box-sizing: border-box; }
                        body {
                            font-family: Arial, sans-serif;
                            margin: 0;
                            color: #0f172a;
                            background: #ffffff;
                        }
                        .page {
                            width: 210mm;
                            min-height: 297mm;
                            margin: 0 auto;
                            padding: 18mm 16mm;
                        }
                        .header {
                            text-align: center;
                            border-bottom: 3px solid #0f172a;
                            padding-bottom: 12px;
                            margin-bottom: 22px;
                        }
                        .header h1 {
                            margin: 0;
                            font-size: 18px;
                            text-transform: uppercase;
                        }
                        .header p {
                            margin: 5px 0 0;
                            font-size: 12px;
                        }
                        .grid {
                            display: grid;
                            grid-template-columns: 180px 1fr;
                            gap: 8px 14px;
                            margin-bottom: 20px;
                            font-size: 12px;
                        }
                        .label { font-weight: 700; }
                        table {
                            width: 100%;
                            border-collapse: collapse;
                            margin-top: 14px;
                            font-size: 11px;
                        }
                        th, td {
                            border: 1px solid #475569;
                            padding: 8px;
                            vertical-align: top;
                        }
                        th {
                            background: #e2e8f0;
                            text-align: center;
                        }
                        .amount { text-align: right; white-space: nowrap; }
                        .signature {
                            margin-top: 48px;
                            display: flex;
                            justify-content: flex-end;
                        }
                        .signature-box {
                            width: 260px;
                            text-align: center;
                            font-size: 12px;
                        }
                        .signature-space { height: 75px; }
                        .actions {
                            position: fixed;
                            top: 16px;
                            right: 16px;
                        }
                        .actions button {
                            border: 0;
                            border-radius: 8px;
                            background: #0f172a;
                            color: white;
                            padding: 10px 16px;
                            cursor: pointer;
                        }
                        @media print {
                            .actions { display: none; }
                            .page {
                                width: auto;
                                min-height: auto;
                                margin: 0;
                                padding: 0;
                            }
                        }
                    </style>
                </head>
                <body>
                    <div class="actions">
                        <button onclick="window.print()">Cetak</button>
                    </div>
                    <div class="page">
                        ${body}
                    </div>
                </body>
            </html>
        `);
        printWindow.document.close();
        printWindow.focus();
    }

    function printTravelReport(row: DalkotRow) {
        const officerRows = row.officers
            .map(
                (officer, index) => `
                    <tr>
                        <td style="text-align:center">${index + 1}</td>
                        <td>${escapeHtml(officer.name)}</td>
                        <td>${escapeHtml(officer.nip || '-')}</td>
                        <td>${escapeHtml(officer.position || '-')}</td>
                        <td class="amount">${escapeHtml(formatRupiah(officer.amount))}</td>
                    </tr>
                `
            )
            .join('');

        openPrintWindow(
            `Laporan Perjalanan Dalkot ${row.businessId}`,
            `
                <div class="header">
                    <h1>Laporan Perjalanan Dinas Dalam Kota</h1>
                    <p>Protokol Kementerian Ketenagakerjaan Republik Indonesia</p>
                </div>

                <div class="grid">
                    <div class="label">Nomor Data</div>
                    <div>${escapeHtml(row.businessId)}</div>

                    <div class="label">Tanggal Pelaksanaan</div>
                    <div>${escapeHtml(formatDate(row.serviceDate))}</div>

                    <div class="label">Jenis Dalkot</div>
                    <div>${escapeHtml(formatType(row))}</div>

                    <div class="label">Kategori</div>
                    <div>${escapeHtml(
                        row.dalkotType === 'KEBIJAKAN_PROTOKOL'
                            ? formatPolicyCategory(row.policyCategory)
                            : row.category
                    )}</div>

                    <div class="label">Pejabat Didampingi</div>
                    <div>${escapeHtml(row.accompaniedOfficial)}</div>

                    <div class="label">Nama Kegiatan</div>
                    <div>${escapeHtml(row.activityName)}</div>

                    <div class="label">Lokasi</div>
                    <div>${escapeHtml(row.location)}</div>

                    <div class="label">Isi Laporan</div>
                    <div>${escapeHtml(row.reportContent || '-')}</div>
                </div>

                <table>
                    <thead>
                        <tr>
                            <th style="width:45px">No</th>
                            <th>Nama Petugas</th>
                            <th>NIP atau NRP</th>
                            <th>Jabatan</th>
                            <th style="width:140px">Biaya</th>
                        </tr>
                    </thead>
                    <tbody>
                        ${officerRows || `
                            <tr>
                                <td colspan="5" style="text-align:center">Belum ada petugas.</td>
                            </tr>
                        `}
                    </tbody>
                    <tfoot>
                        <tr>
                            <th colspan="4" style="text-align:right">Total</th>
                            <th class="amount">${escapeHtml(formatRupiah(row.totalAmount))}</th>
                        </tr>
                    </tfoot>
                </table>
            `
        );
    }

    function printDpr(row: DalkotRow, officer: DalkotOfficer) {
        openPrintWindow(
            `DPR ${officer.name}`,
            `
                <div class="header">
                    <h1>Daftar Pengeluaran Riil</h1>
                    <p>Perjalanan Dinas Dalam Kota</p>
                </div>

                <div class="grid">
                    <div class="label">Nomor Surat</div>
                    <div>${escapeHtml(row.businessId)}</div>

                    <div class="label">Nama Petugas</div>
                    <div>${escapeHtml(officer.name)}</div>

                    <div class="label">NIP atau NRP</div>
                    <div>${escapeHtml(officer.nip || '-')}</div>

                    <div class="label">Jabatan</div>
                    <div>${escapeHtml(officer.position || '-')}</div>

                    <div class="label">Tanggal Pelaksanaan</div>
                    <div>${escapeHtml(formatDate(row.serviceDate))}</div>

                    <div class="label">Nama Kegiatan</div>
                    <div>${escapeHtml(row.activityName)}</div>

                    <div class="label">Lokasi</div>
                    <div>${escapeHtml(row.location)}</div>
                </div>

                <table>
                    <thead>
                        <tr>
                            <th style="width:55px">No</th>
                            <th>Uraian Pengeluaran</th>
                            <th style="width:160px">Jumlah</th>
                        </tr>
                    </thead>
                    <tbody>
                        <tr>
                            <td style="text-align:center">1</td>
                            <td>Transportasi perjalanan dinas dalam kota</td>
                            <td class="amount">${escapeHtml(formatRupiah(officer.amount))}</td>
                        </tr>
                    </tbody>
                    <tfoot>
                        <tr>
                            <th colspan="2" style="text-align:right">Jumlah</th>
                            <th class="amount">${escapeHtml(formatRupiah(officer.amount))}</th>
                        </tr>
                    </tfoot>
                </table>

                <div class="signature">
                    <div class="signature-box">
                        <p>Yang menerima,</p>
                        <div class="signature-space"></div>
                        <strong>${escapeHtml(officer.name)}</strong>
                        <p>${escapeHtml(officer.nip || '')}</p>
                    </div>
                </div>
            `
        );
    }
</script>

<main class="space-y-6 pb-20">
    <section

    >
        <div

        ></div>
        <div

        ></div>

        <div
            class="relative z-10 flex flex-col gap-6 lg:flex-row lg:items-end lg:justify-between"
        >
            <div class="max-w-3xl">
                <p
                    class="text-xs font-bold uppercase tracking-[0.2em] text-sky-200"
                >
                    Perjalanan Dinas Dalam Kota
                </p>

                <h1 class="mt-3 text-3xl font-black sm:text-4xl">
                    Daftar Dalkot
                </h1>

                <p class="mt-3 text-sm leading-6 text-slate-200 sm:text-base">
                    Tinjau data perjalanan, total biaya, status, rincian petugas,
                    laporan perjalanan, dan Daftar Pengeluaran Riil.
                </p>
            </div>

            <button
                type="button"
                class="inline-flex items-center justify-center gap-2 rounded-2xl bg-amber-400 px-5 py-3 text-sm font-bold text-slate-950 shadow-lg transition hover:bg-amber-300"
                on:click={() => goto('/dashboard/dalkot/pengajuan')}
            >
                <Plus size={18} />
                Tambah Pengajuan
            </button>
        </div>
    </section>

    {#if loadError}
        <section
            class="rounded-2xl border border-red-200 bg-red-50 p-5 text-sm text-red-700"
        >
            {loadError}

            <button
                type="button"
                class="ml-2 font-bold underline"
                on:click={loadDalkot}
            >
                Muat ulang
            </button>
        </section>
    {/if}

    <section class="grid grid-cols-1 gap-4 md:grid-cols-2 xl:grid-cols-4">
        <article class="rounded-2xl border border-slate-200 bg-white p-5 shadow-sm">
            <div class="flex items-center justify-between">
                <div>
                    <p class="text-xs font-bold uppercase tracking-wide text-slate-400">
                        Total Pengajuan
                    </p>
                    <p class="mt-2 text-2xl font-black text-slate-900">
                        {summary.totalRecords}
                    </p>
                </div>

                <div
                    class="flex h-12 w-12 items-center justify-center rounded-2xl bg-blue-50 text-blue-700"
                >
                    <FileText size={22} />
                </div>
            </div>
        </article>

        <article class="rounded-2xl border border-emerald-200 bg-white p-5 shadow-sm">
            <div class="flex items-center justify-between">
                <div>
                    <p class="text-xs font-bold uppercase tracking-wide text-slate-400">
                        Total Petugas
                    </p>
                    <p class="mt-2 text-2xl font-black text-emerald-700">
                        {summary.totalOfficers}
                    </p>
                </div>

                <div
                    class="flex h-12 w-12 items-center justify-center rounded-2xl bg-emerald-50 text-emerald-700"
                >
                    <Users size={22} />
                </div>
            </div>
        </article>

        <article class="rounded-2xl border border-amber-200 bg-white p-5 shadow-sm">
            <div class="flex items-center justify-between">
                <div>
                    <p class="text-xs font-bold uppercase tracking-wide text-slate-400">
                        Total Biaya
                    </p>
                    <p class="mt-2 text-xl font-black text-amber-700">
                        {formatRupiah(summary.totalAmount)}
                    </p>
                </div>

                <div
                    class="flex h-12 w-12 items-center justify-center rounded-2xl bg-amber-50 text-amber-700"
                >
                    <CircleDollarSign size={22} />
                </div>
            </div>
        </article>

        <article class="rounded-2xl border border-violet-200 bg-white p-5 shadow-sm">
            <div class="flex items-center justify-between">
                <div>
                    <p class="text-xs font-bold uppercase tracking-wide text-slate-400">
                        Menunggu Proses
                    </p>
                    <p class="mt-2 text-2xl font-black text-violet-700">
                        {summary.pendingRecords}
                    </p>
                </div>

                <div
                    class="flex h-12 w-12 items-center justify-center rounded-2xl bg-violet-50 text-violet-700"
                >
                    <CalendarDays size={22} />
                </div>
            </div>
        </article>
    </section>

    <section
        class="overflow-hidden rounded-3xl border border-slate-200 bg-white shadow-sm"
    >
        <div
            class="flex flex-col gap-4 border-b border-slate-200 p-5 lg:flex-row lg:items-center lg:justify-between sm:p-6"
        >
            <div>
                <p class="text-xs font-bold uppercase tracking-[0.2em] text-slate-400">
                    Data Perjalanan
                </p>

                <h2 class="mt-2 text-xl font-black text-slate-900">
                    Daftar Perjalanan Dinas Dalam Kota
                </h2>

                <p class="mt-1 text-sm text-slate-500">
                    Gunakan rincian petugas untuk melihat biaya per orang dan mencetak DPR.
                </p>
            </div>

            <div class="rounded-xl bg-slate-900 px-4 py-2 text-xs font-bold text-white">
                {filteredRows.length} data
            </div>
        </div>

        <div
            class="grid grid-cols-1 gap-3 border-b border-slate-200 bg-slate-50/70 p-5 md:grid-cols-2 xl:grid-cols-5 sm:p-6"
        >
            <div class="xl:col-span-2">
                <label
                    for="dalkotSearch"
                    class="mb-2 block text-xs font-bold uppercase tracking-wide text-slate-500"
                >
                    Pencarian
                </label>

                <div class="relative">
                    <Search
                        size={17}
                        class="absolute left-4 top-1/2 -translate-y-1/2 text-slate-400"
                    />

                    <input
                        id="dalkotSearch"
                        bind:value={searchQuery}
                        type="search"
                        placeholder="Cari kegiatan, lokasi, pejabat, atau petugas..."
                        class="w-full rounded-xl border border-slate-300 bg-white py-3 pl-11 pr-4 text-sm text-slate-700 outline-none focus:ring-2 focus:ring-slate-800"
                    />
                </div>
            </div>

            <div>
                <label
                    for="dalkotType"
                    class="mb-2 block text-xs font-bold uppercase tracking-wide text-slate-500"
                >
                    Jenis
                </label>

                <select
                    id="dalkotType"
                    bind:value={typeFilter}
                    class="w-full rounded-xl border border-slate-300 bg-white px-4 py-3 text-sm text-slate-700 outline-none focus:ring-2 focus:ring-slate-800"
                >
                    <option value="">Semua Jenis</option>
                    <option value="SPJ_RILL">SPJ RILL</option>
                    <option value="KEBIJAKAN_PROTOKOL">Kebijakan Protokol</option>
                </select>
            </div>

            <div>
                <label
                    for="dalkotMonth"
                    class="mb-2 block text-xs font-bold uppercase tracking-wide text-slate-500"
                >
                    Bulan
                </label>

                <select
                    id="dalkotMonth"
                    bind:value={monthFilter}
                    class="w-full rounded-xl border border-slate-300 bg-white px-4 py-3 text-sm text-slate-700 outline-none focus:ring-2 focus:ring-slate-800"
                >
                    <option value="">Semua Bulan</option>

                    {#each monthOptions as month}
                        <option value={month[0]}>
                            {month[1]}
                        </option>
                    {/each}
                </select>
            </div>

            <div>
                <label
                    for="dalkotYear"
                    class="mb-2 block text-xs font-bold uppercase tracking-wide text-slate-500"
                >
                    Tahun
                </label>

                <div class="flex gap-2">
                    <select
                        id="dalkotYear"
                        bind:value={yearFilter}
                        class="min-w-0 flex-1 rounded-xl border border-slate-300 bg-white px-4 py-3 text-sm text-slate-700 outline-none focus:ring-2 focus:ring-slate-800"
                    >
                        <option value="">Semua Tahun</option>

                        {#each availableYears as year}
                            <option value={year}>
                                {year}
                            </option>
                        {/each}
                    </select>

                    <button
                        type="button"
                        class="flex h-12 w-12 shrink-0 items-center justify-center rounded-xl border border-slate-300 bg-white text-slate-600 transition hover:bg-slate-100"
                        title="Reset filter"
                        aria-label="Reset filter"
                        on:click={resetFilters}
                    >
                        <RefreshCcw size={17} />
                    </button>
                </div>
            </div>
        </div>
